Arguments (pass arguments as '<name>:=<value>'):

    'rviz':
        Open RViz
        (default: 'true')

    'use_usbl':
        Fuse two-way USBL horizontal position
        (default: 'true')

    'record_path':
        Publish EKF Path and CSV
        (default: 'true')

    'csv_path':
        CSV file; empty creates a timestamped file in ~/.ros/baseline_robot/navigation
        (default: '')

    'arg_vehicle_name':
        no description given
        (default: 'baseline_robot')

    'arg_world_frame':
        no description given
        (default: 'baseline_robot/world')

    'arg_imu_frame':
        no description given
        (default: 'baseline_robot/imu_sf')
